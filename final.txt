#!/usr/sbin/nft -f
flush ruleset

define WAN     = "eth0"
define LAN     = "eth1"
define BACKEND = 192.168.10.10    # internal host receiving ports 80/8080

table ip router {

    set web_ports {
        type inet_service
        elements = { 80, 8080 }
    }

    chain input {
        type filter hook input priority filter; policy drop;
        ct state established,related accept
        ct state invalid drop
        iif lo accept
        icmp type echo-request accept
        tcp dport 22 accept          # keep SSH, otherwise you lock yourself out
    }

    chain forward {
        type filter hook forward priority filter; policy drop;
        ct state established,related accept
        iifname $LAN oifname $WAN accept
        iifname $WAN oifname $LAN ip daddr $BACKEND tcp dport @web_ports ct status dnat accept
    }

    chain prerouting {
        type nat hook prerouting priority dstnat; policy accept;
        iifname $WAN tcp dport @web_ports dnat to $BACKEND
    }

    chain postrouting {
        type nat hook postrouting priority srcnat; policy accept;
        oifname $WAN masquerade
    }
}

#!/usr/sbin/nft -f

flush ruleset

table inet filter {
    chain input {
        type filter hook input priority filter; policy drop;

        iif "lo" accept
        ct state established,related accept

        # Allow SSH and ICMP
        tcp dport 22 accept
        icmp type echo-request accept

        # Allow WireGuard inbound on WAN
        udp dport { 51820, 51821 } accept
    }

    chain forward {
        type filter hook forward priority filter; policy drop;

        ct state established,related accept

        # Allow cross-site transit between HQ, CP, and CHQ
        iifname "wg0" oifname "wg1" accept
        iifname "wg1" oifname "wg0" accept

        # Allow LAN traffic to/from VPN tunnels
        ip saddr 10.101.50.0/24 oifname { "wg0", "wg1" } accept
        iifname { "wg0", "wg1" } ip daddr 10.101.50.0/24 accept

        # Allow LAN/Router access to WAN Repository (203.0.113.20)
        ip daddr 203.0.113.20 accept
    }

    chain output {
        type filter hook output priority filter; policy accept;
    }
}

table ip nat {
    chain postrouting {
        type nat hook postrouting priority srcnat; policy accept;

        # Source NAT for Linux Repository updates over WAN
        ip daddr 203.0.113.20 oif "<WAN_IF>" masquerade
    }
}
